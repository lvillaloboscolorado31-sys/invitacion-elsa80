    <script>
        // Array de fotos de Elsa
        const photos = [
            "Elsa_fondo_dorado_foto_1.jpg",
            "Elsa_fondo_dorado_foto_2.jpg",
            "Elsa_fondo_dorado_foto_3.jpg"
        ];
        let currentPhotoIndex = 0;

        function changePhoto(index) {
            currentPhotoIndex = index;
            const imgEl = document.getElementById("carousel-img");
            imgEl.style.opacity = "0.2";
            setTimeout(() => {
                imgEl.src = photos[currentPhotoIndex];
                imgEl.style.opacity = "1";
            }, 300);

            for (let i = 0; i < 3; i++) {
                const dot = document.getElementById(`dot-${i}`);
                if (i === index) {
                    dot.className = "w-2.5 h-2.5 rounded-full bg-white/90 transition-all scale-125";
                } else {
                    dot.className = "w-2.5 h-2.5 rounded-full bg-white/40 transition-all";
                }
            }
        }

        // Cambio automático de fotos cada 4 segundos
        setInterval(() => {
            let nextIndex = (currentPhotoIndex + 1) % photos.length;
            changePhoto(nextIndex);
        }, 4000);

        // Control de Música
        const bgMusic = document.getElementById("bg-music");
        const musicBtn = document.getElementById("music-btn");
        const musicIcon = document.getElementById("music-icon");
        let isPlaying = false;

        function abrirInvitacion() {
            document.getElementById("envelope-screen").classList.add("open-envelope");
            document.getElementById("music-control").classList.remove("hidden");
            
            // Iniciar música
            bgMusic.play().then(() => {
                isPlaying = true;
                musicIcon.className = "fa-solid fa-compact-disc text-lg animate-spin";
            }).catch(e => {
                console.log("Autoplay bloqueado por el navegador:", e);
            });
        }

        function toggleMusic() {
            if (isPlaying) {
                bgMusic.pause();
                isPlaying = false;
                musicIcon.className = "fa-solid fa-music text-lg";
            } else {
                bgMusic.play();
                isPlaying = true;
                musicIcon.className = "fa-solid fa-compact-disc text-lg animate-spin";
            }
        }

        // Cuenta Regresiva Real (31 de Octubre 2026, 7:00 PM)
        const eventDate = new Date("Oct 31, 2026 19:00:00").getTime();

        const countdownInterval = setInterval(() => {
            const now = new Date().getTime();
            const distance = eventDate - now;

            if (distance < 0) {
                clearInterval(countdownInterval);
                document.getElementById("days").innerText = "00";
                document.getElementById("hours").innerText = "00";
                document.getElementById("minutes").innerText = "00";
                document.getElementById("seconds").innerText = "00";
                return;
            }

            const days = Math.floor(distance / (1000 * 60 * 60 * 24));
            const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
            const seconds = Math.floor((distance % (1000 * 60)) / 1000);

            document.getElementById("days").innerText = days < 10 ? '0' + days : days;
            document.getElementById("hours").innerText = hours < 10 ? '0' + hours : hours;
            document.getElementById("minutes").innerText = minutes < 10 ? '0' + minutes : minutes;
            document.getElementById("seconds").innerText = seconds < 10 ? '0' + seconds : seconds;
        }, 1000);

        // Generador de Destellos en el Fondo
        const sparklesContainer = document.getElementById("sparkles-container");
        for (let i = 0; i < 35; i++) {
            const sparkle = document.createElement("div");
            sparkle.className = "sparkle";
            const size = Math.random() * 4 + 2;
            sparkle.style.width = `${size}px`;
            sparkle.style.height = `${size}px`;
            sparkle.style.left = `${Math.random() * 100}%`;
            sparkle.style.animationDuration = `${Math.random() * 8 + 5}s`;
            sparkle.style.animationDelay = `${Math.random() * 5}s`;
            sparklesContainer.appendChild(sparkle);
        }
    </script>
